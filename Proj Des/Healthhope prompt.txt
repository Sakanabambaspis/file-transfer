def extract_product_id(url):
    """
    Extracts the product ID from a DM.de product URL.
    Returns the ID without the leading 'p' or None if not found.
    """
    # Pattern to match the product ID (p followed by numbers at the end before .html)
    pattern = r'p(\d+)\.html$'
    
    # Search for the pattern in the URL
    match = re.search(pattern, url)
    
    if match:
        print('~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~')
        print(f"Original URL: {url}")
        print(f"Extracted Product ID: {match.group(1)}")
        return match.group(1)  # Return the digits without the 'p'
    print("No product ID found")
    return None

import requests
from concurrent.futures import ThreadPoolExecutor
import regex

# Function to check if a URL is allowed by robots.txt
def is_allowed_by_robots(url, user_agent='*'):
    parsed_url = urlparse(url)
    base_url = f"{parsed_url.scheme}://{parsed_url.netloc}"
    robots_url = urljoin(base_url, '/robots.txt')
    
    try:
        response = requests.get(robots_url, timeout=10)
        response.raise_for_status()
        robots_content = response.text.splitlines()
        
        allow_rules = []
        disallow_rules = []
        
        for line in robots_content:
            line = line.strip()
            if line.startswith('User-agent:'):
                current_user_agent = line.split(': ')[1].strip()
            elif line.startswith('Allow:') and current_user_agent == user_agent:
                allow_rules.append(line.split(': ')[1].strip())
            elif line.startswith('Disallow:') and current_user_agent == user_agent:
                disallow_rules.append(line.split(': ')[1].strip())
        
        # Check against disallow rules
        for rule in disallow_rules:
            if rule.endswith('$'):
                if url == urljoin(base_url, rule[:-1]):
                    return False
            elif rule.endswith('*'):
                if url.startswith(urljoin(base_url, rule[:-1])):
                    return False
            else:
                if url == urljoin(base_url, rule):
                    return False
        
        # Check against allow rules
        for rule in allow_rules:
            if rule.endswith('$'):
                if url == urljoin(base_url, rule[:-1]):
                    return True
            elif rule.endswith('*'):
                if url.startswith(urljoin(base_url, rule[:-1])):
                    return True
            else:
                if url == urljoin(base_url, rule):
                    return True
        
        # Default to disallow if no specific rule matches
        return len(disallow_rules) == 0
    
    except requests.exceptions.RequestException as e:
        print(f"Error fetching robots.txt: {e}")
        return False

import os
import pandas as pd

def remove_duplicates_from_csv_folder(folder_path, overwrite=True):
    """
    遍历指定文件夹中的所有 .csv 文件，并删除重复行。
    
    参数:
        folder_path (str): 包含 CSV 文件的文件夹路径
        overwrite (bool): 是否覆盖原始文件，默认是；否则保存为 _deduplicated.csv
    """
    for filename in os.listdir(folder_path):
        if filename.endswith('.csv'):
            file_path = os.path.join(folder_path, filename)
            
            try:
                # 使用 pandas 读取 CSV
                df = pd.read_csv(file_path)

                # 删除重复行（保留第一次出现）
                df.drop_duplicates(inplace=True)

                # 构造输出路径
                if overwrite:
                    output_path = file_path
                else:
                    name, ext = os.path.splitext(filename)
                    output_path = os.path.join(folder_path, f"{name}_deduplicated{ext}")

                # 写回 CSV 文件
                df.to_csv(output_path, index=False)
                print(f"Processed: {filename}")
            except Exception as e:
                print(f"Error processing {filename}: {e}")

def add_base_url(input_file, output_file, base_url):
    with open(input_file, 'r', encoding='utf-8') as infile:
        lines = infile.readlines()

    # 去除每行的换行符，并拼接基础URL
    full_urls = [base_url + line.strip() + '\n' for line in lines]

    with open(output_file, 'w', encoding='utf-8') as outfile:
        outfile.writelines(full_urls)

import requests

def test_anti_crawling(url):
    # Define a generic User-Agent
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/136.0.0.0 Safari/537.36 Edg/136.0.0.0"
    }

    try:
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()

        print(f"Status Code: {response.status_code}")
        print(f"Response Headers: {response.headers}")

        # Check for specific headers indicating rate limiting
        if 'X-RateLimit-Limit' in response.headers or 'X-RateLimit-Remaining' in response.headers:
            print("Rate limiting detected.")

        # Check for CAPTCHA-like content
        if 'captcha' in response.text.lower():
            print("CAPTCHA detected.")

        # Check for robots.txt restrictions
        robots_url = f"https://{url.split('//')[1].split('/')[0]}/robots.txt"
        robots_response = requests.get(robots_url, headers=headers, timeout=10)
        if robots_response.ok and 'Disallow:' in robots_response.text:
            print("Disallowed paths found in robots.txt:")
            print(robots_response.text)

    except requests.exceptions.HTTPError as http_err:
        print(f"HTTP error occurred: {http_err}")
    except requests.exceptions.RequestException as err:
        print(f"Other error occurred: {err}")

import re
import json

def process_raw_dump(input_data):
    processed_list = []
    
    # Anchor for the start of Requirements: "1、2027届" or "1、2026届"
    req_anchor = r"1、202\d届"

    for item in input_data:
        title = item.get("title", "")
        raw_text = item.get("raw_text", "")
        link = item.get("link", "")
        
        # 1. Locate the Requirements section
        req_match = re.search(req_anchor, raw_text)
        
        if req_match:
            # Requirements is everything from the anchor to the end
            requirements = raw_text[req_match.start():].strip()
            
            # 2. Extract Responsibilities from the text BEFORE the anchor
            # We strip out the Title and the Team Intro
            pre_req_text = raw_text[:req_match.start()]
            
            # Bytedance JDs usually have a double newline before the numbered list of duties
            # We look for the first "1、" that isn't the graduation requirement
            resp_match = re.search(r"\n\n1、(.*)", pre_req_text, re.S)
            
            if resp_match:
                responsibilities = "1、" + resp_match.group(1).strip()
            else:
                # Fallback if the \n\n1、 pattern fails
                responsibilities = pre_req_text.split("团队介绍：")[-1].strip()
        else:
            requirements = "Manual check required"
            responsibilities = raw_text

        processed_list.append({
            "title": title,
            "responsibilities": responsibilities,
            "requirements": requirements,
            "link": link
        })
        
    return processed_list

# Usage
with open('bytedance_raw_dump.json', 'r', encoding='utf-8') as f:

cleaned = process_raw_dump(raw_dump)

with open('bytedance_cleaned_dump.json', "w", encoding='utf-8') as json_file:

import re
import json

def filter_dump(input_data):
    filtered_list = []

    """
    1. Filter for undergrad
    2. Filter for Python, PyTorch, NLP, 微调, 数据处理
    """
    
    req_anchor1 = r"本科"
    req_anchor2 = r"Python|PyTorch|NLP|微调|数据处理"

    for item in input_data:
        title = item.get("title", "")
        responsibilities = item.get("responsibilities", "")
        requirements = item.get("requirements", "")
        
        # 1. Locate the Requirements section
        req1_match = re.search(req_anchor1, requirements)
        req2_match = re.search(req_anchor2, requirements) or re.search(req_anchor2, responsibilities)
        
        if req1_match and req2_match:
            filtered_list.append(item)
        
    return filtered_list

# Usage
